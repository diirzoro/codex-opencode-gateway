const dashboardNode=(tag,text='',className='')=>textElement(tag,text,className);
function dashboardAction(text,handler){const button=dashboardNode('button',text,'button ghost small');button.type='button';button.onclick=async()=>{button.disabled=true;try{await handler()}catch(error){toast(error.message)}finally{button.disabled=false}};return button}
function dashboardEmpty(root,en,ar){root.append(dashboardNode('p',dashboardText(en,ar),'dashboard-empty'))}
function dashboardStat(root,label,value,note=''){const card=dashboardNode('article');card.append(dashboardNode('small',label),dashboardNode('h3',String(value)));if(note)card.append(dashboardNode('p',note));root.append(card)}
function dashboardRow(root,title,detail,actions=[]){const row=dashboardNode('article','','dashboard-project-row'),copy=dashboardNode('div');copy.append(dashboardNode('strong',title),dashboardNode('small',detail));const controls=dashboardNode('div','','manage-actions');controls.append(...actions);row.append(copy,controls);root.append(row)}
async function openDashboardWorkspace(workspace,session=null){const project=clientDashboardData.projects.find(p=>p.id===workspace.project_id);if(!project)throw new Error('Project no longer available');await openWorkspace(project,workspace);if(session)await selectSession(session)}
function renderDashboardProjects(root,projects){root.replaceChildren();if(!projects.length)return dashboardEmpty(root,'No projects yet. Create your first project.','لا توجد مشاريع بعد. أنشئ مشروعك الأول.');for(const project of projects){const actions=project.workspaces.map(workspace=>dashboardAction(dashboardText('Open workspace','فتح مساحة العمل'),()=>openDashboardWorkspace(workspace)));dashboardRow(root,project.name,[project.source_type,project.repository,project.remote_branch||project.branch].filter(Boolean).join(' · '),actions)}}
function renderDashboardSessions(root,data){root.replaceChildren();if(!data.recent_sessions.length)dashboardEmpty(root,'No conversations yet. Open a workspace to start a session.','لا توجد محادثات بعد. افتح مساحة عمل لبدء جلسة.');for(const session of data.recent_sessions){const workspace=data.projects.flatMap(p=>p.workspaces).find(w=>w.id===session.workspace_id);dashboardRow(root,session.title,session.status+' · '+new Date(session.created_at).toLocaleString(),workspace?[dashboardAction(dashboardText('Open session','فتح الجلسة'),()=>openDashboardWorkspace(workspace,session))]:[])}}
async function loadClientDashboard(){
  $('#clientSummary').replaceChildren(dashboardNode('p',dashboardText('Loading your dashboard…','جارٍ تحميل لوحتك…')));
  const data=await api('/api/dashboard');clientDashboardData=data;
  if(!currentUser||!$('#clientPage').classList.contains('active'))return;
  const summary=$('#clientSummary');summary.replaceChildren();
  dashboardStat(summary,dashboardText('Trial days remaining','أيام التجربة المتبقية'),data.profile.trial_remaining_days,dashboardText('Ends ','تنتهي ')+new Date(data.profile.trial_ends_at).toLocaleDateString());
  dashboardStat(summary,dashboardText('Account & plan','الحساب والخطة'),data.profile.status,data.plan?.name||dashboardText('No paid plan assigned','لا توجد خطة مدفوعة معيّنة'));
  dashboardStat(summary,'GitHub',data.github.connected?dashboardText('Connected','متصل'):dashboardText('Not connected','غير متصل'),data.github.account_login||dashboardText('Connect your own GitHub account','اربط حساب GitHub الخاص بك'));
  dashboardStat(summary,'OpenCode',data.runtime.healthy?dashboardText('Available','متاح'):dashboardText('Not connected','غير متصل'),data.runtime.mode);
  dashboardStat(summary,dashboardText('Saved provider credentials','بيانات اتصال المزود المحفوظة'),data.counts.credentials,dashboardText('Per workspace · not a provider health check','لكل مساحة عمل · ليست اختبارًا لصحة الاتصال'));
  dashboardStat(summary,dashboardText('Projects / workspaces / sessions','المشاريع / المساحات / الجلسات'),`${data.counts.projects} / ${data.counts.workspaces} / ${data.counts.sessions}`);
  renderDashboardProjects($('#clientRecentProjects'),data.projects.slice(0,5));renderDashboardProjects($('#clientProjectsList'),data.projects);
  renderDashboardSessions($('#clientRecentSessions'),data);renderDashboardSessions($('#clientSessionsList'),data);
  const workspaces=$('#clientRecentWorkspaces');workspaces.replaceChildren();
  for(const workspace of data.recent_workspaces){const project=data.projects.find(p=>p.id===workspace.project_id);dashboardRow(workspaces,project?.name||dashboardText('Workspace','مساحة عمل'),workspace.status+' · '+workspace.session_count+' '+dashboardText('sessions','جلسات'),[dashboardAction(dashboardText('Open workspace','فتح مساحة العمل'),()=>openDashboardWorkspace(workspace))])}
  if(!data.recent_workspaces.length)dashboardEmpty(workspaces,'No workspaces yet.','لا توجد مساحات عمل بعد.');
  setClientPanel('dashboard');
}
